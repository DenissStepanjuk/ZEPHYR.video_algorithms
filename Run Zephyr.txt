* Windows PowerShell:
# Вывести список USB-устройств, обнаруженных операционной системой Windows.
usbipd list

# Выполняет привязку USB-устройства к службе USB/IP в Windows.
usbipd bind --busid 1-9

# Передать USB-устройство с идентификатором 1-9 в среду WSL2 Ubuntu.
usbipd attach --wsl --busid 1-9

# Собрать проект
ESP32 S3 CAM N16R8:
west build -p always -b esp32s3_eye/esp32s3/procpu -- -DDTC_OVERLAY_FILE=boards/esp32s3_eye.overlay -DEXTRA_CONF_FILE=boards/esp32s3_eye.conf

west flash

screen /dev/ttyACM0 115200

mini_stm32h743:
west build -p always -b mini_stm32h743 -- -DDTC_OVERLAY_FILE=boards/mini_stm32h743.overlay -DEXTRA_CONF_FILE=boards/mini_stm32h743.conf

sudo -E /home/stepanjuk/zephyrproject/.venv/bin/west flash

screen /dev/ttyACM0 115200
